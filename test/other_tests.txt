@testset "pead_params" begin
    @testset "Read params with CLI available" begin
        #  mock read_safe_yaml to return a dict.
        @mock dso_cli_available()=true begin
        @mock run_dso(command) = nothing begin
        @mock read_safe_yaml(params_file)=Dict("a" => 1) begin

            current_directory = pwd()
            cd(tmp)            

            params_obj = read_params()
            params_dict = read_params(return_dict = true)

            cd(current_directory)

            @test params_obj isa DsoParams
            @test params_obj.a == 1
            @test haskey(params_dict, 'a') == true
            @test get(params_dict,"a",2) == 1
        end; end; end
    end

    @testset "Read params without CLI available" begin
        mktempdir() do tmp
            # Mock dso cli
            @mock dso_cli_available()=false begin
            @mock read_safe_yaml(params_file)=Dict("a" => 1) begin
                @test_throws ErrorException read_params()
            end; end
    end
end
